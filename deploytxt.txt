## 📁 Файл 1: `deploy.sh`

Создай в корне проекта файл `deploy.sh` и вставь этот код:

```bash
#!/bin/bash

# ============================================================================
# POS System - Автоматическое развертывание на Debian/Ubuntu VPS
# Стек: Django 5.0 + React (Vite) + PostgreSQL + Redis + Nginx + Daphne
# ============================================================================

set -e  # Остановка при ошибке

# Цвета для вывода
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m'

# Проверка, что скрипт запущен с правами root
if [ "$EUID" -ne 0 ]; then
    echo -e "${RED}❌ Запустите скрипт с правами root: sudo ./deploy.sh${NC}"
    exit 1
fi

# Получаем реального пользователя (не root)
REAL_USER=${SUDO_USER:-$USER}
REAL_HOME=$(eval echo ~$REAL_USER)

echo -e "${GREEN}"
echo "╔════════════════════════════════════════════════════════════╗"
echo "║   🚀 POS System - Автоматическое развертывание            ║"
echo "║   Стек: Django + React + PostgreSQL + Redis + Nginx       ║"
echo "╚════════════════════════════════════════════════════════════╝"
echo -e "${NC}"

# ----------------------------------------------------------------------------
# ШАГ 1: Загрузка переменных окружения
# ----------------------------------------------------------------------------
echo -e "${YELLOW}📋 Шаг 1/11: Загрузка конфигурации...${NC}"

if [ ! -f .env ]; then
    echo -e "${RED}❌ Файл .env не найден!${NC}"
    echo -e "${YELLOW}💡 Скопируйте .env.example в .env и заполните его:${NC}"
    echo -e "   cp .env.example .env"
    echo -e "   nano .env"
    exit 1
fi

# Загружаем переменные из .env
set -a
source .env
set +a

# Проверка обязательных переменных
if [ -z "$DB_PASSWORD" ] || [ -z "$DJANGO_SECRET_KEY" ]; then
    echo -e "${RED}❌ В .env не заполнены обязательные поля (DB_PASSWORD, DJANGO_SECRET_KEY)${NC}"
    exit 1
fi

# Устанавливаем значения по умолчанию
PROJECT_DIR=${PROJECT_DIR:-/opt/pos_system}
DOMAIN=${DOMAIN:-localhost}
DB_NAME=${DB_NAME:-pos_db}
DB_USER=${DB_USER:-pos_user}
ADMIN_USERNAME=${ADMIN_USERNAME:-admin}
ADMIN_EMAIL=${ADMIN_EMAIL:-admin@example.com}
ADMIN_PASSWORD=${ADMIN_PASSWORD:-admin123}

echo -e "${GREEN}✅ Конфигурация загружена${NC}"
echo -e "   📁 Проект: ${PROJECT_DIR}"
echo -e "   🌐 Домен: ${DOMAIN}"
echo -e "   🗄️  БД: ${DB_NAME}"

# ----------------------------------------------------------------------------
# ШАГ 2: Обновление системы
# ----------------------------------------------------------------------------
echo -e "${YELLOW}📦 Шаг 2/11: Обновление системы...${NC}"
apt update -y
apt upgrade -y
apt install -y \
    python3 python3-pip python3-venv python3-dev \
    postgresql postgresql-contrib \
    redis-server \
    nginx \
    supervisor \
    git curl build-essential \
    ufw

echo -e "${GREEN}✅ Системные пакеты установлены${NC}"

# ----------------------------------------------------------------------------
# ШАГ 3: Установка Node.js 20.x LTS
# ----------------------------------------------------------------------------
echo -e "${YELLOW}🟢 Шаг 3/11: Установка Node.js 20.x LTS...${NC}"

if ! command -v node &> /dev/null; then
    curl -fsSL https://deb.nodesource.com/setup_20.x | bash -
    apt install -y nodejs
fi

echo -e "${GREEN}✅ Node.js $(node -v) установлен${NC}"

# ----------------------------------------------------------------------------
# ШАГ 4: Настройка PostgreSQL
# ----------------------------------------------------------------------------
echo -e "${YELLOW}🐘 Шаг 4/11: Настройка PostgreSQL...${NC}"

systemctl enable postgresql
systemctl start postgresql

# Создаём пользователя и БД (игнорируем ошибки, если уже существуют)
sudo -u postgres psql -tc "SELECT 1 FROM pg_user WHERE usename = '${DB_USER}'" | grep -q 1 || \
    sudo -u postgres psql -c "CREATE USER ${DB_USER} WITH PASSWORD '${DB_PASSWORD}';"

sudo -u postgres psql -tc "SELECT 1 FROM pg_database WHERE datname = '${DB_NAME}'" | grep -q 1 || \
    sudo -u postgres psql -c "CREATE DATABASE ${DB_NAME} OWNER ${DB_USER};"

sudo -u postgres psql -c "GRANT ALL PRIVILEGES ON DATABASE ${DB_NAME} TO ${DB_USER};" 2>/dev/null || true
sudo -u postgres psql -d ${DB_NAME} -c "GRANT ALL ON SCHEMA public TO ${DB_USER};" 2>/dev/null || true

echo -e "${GREEN}✅ PostgreSQL настроен${NC}"

# ----------------------------------------------------------------------------
# ШАГ 5: Настройка Redis
# ----------------------------------------------------------------------------
echo -e "${YELLOW}🔴 Шаг 5/11: Настройка Redis...${NC}"

systemctl enable redis-server
systemctl start redis-server

if redis-cli ping | grep -q "PONG"; then
    echo -e "${GREEN}✅ Redis работает${NC}"
else
    echo -e "${RED}❌ Redis не запустился!${NC}"
    exit 1
fi

# ----------------------------------------------------------------------------
# ШАГ 6: Создание структуры проекта
# ----------------------------------------------------------------------------
echo -e "${YELLOW}📁 Шаг 6/11: Подготовка директории проекта...${NC}"

mkdir -p ${PROJECT_DIR}
chown -R ${REAL_USER}:${REAL_USER} ${PROJECT_DIR}

# Копируем файлы из текущей директории в PROJECT_DIR (если они есть)
if [ -d "core" ] && [ -d "pos" ] && [ -f "manage.py" ]; then
    echo -e "${BLUE}   Копируем файлы проекта...${NC}"
    cp -r core pos manage.py ${PROJECT_DIR}/ 2>/dev/null || true
    [ -d "frontend" ] && cp -r frontend ${PROJECT_DIR}/
    [ -f "requirements.txt" ] && cp requirements.txt ${PROJECT_DIR}/
    [ -f ".env" ] && cp .env ${PROJECT_DIR}/
    echo -e "${GREEN}✅ Файлы скопированы в ${PROJECT_DIR}${NC}"
else
    echo -e "${YELLOW}⚠️  Файлы проекта не найдены в текущей директории${NC}"
    echo -e "${YELLOW}💡 Убедитесь, что вы запустили скрипт из папки с проектом${NC}"
fi

cd ${PROJECT_DIR}

# ----------------------------------------------------------------------------
# ШАГ 7: Python виртуальное окружение
# ----------------------------------------------------------------------------
echo -e "${YELLOW}🐍 Шаг 7/11: Настройка Python окружения...${NC}"

# Создаём venv от имени пользователя
sudo -u ${REAL_USER} python3 -m venv venv
sudo -u ${REAL_USER} bash -c "source venv/bin/activate && pip install --upgrade pip"

# Устанавливаем зависимости с ФИКСИРОВАННЫМИ версиями
# ВАЖНО: redis<5.0.0 из-за бага с RESP3 в связке с channels_redis
sudo -u ${REAL_USER} bash -c "source venv/bin/activate && pip install \
    'Django==5.0.6' \
    'djangorestframework==3.15.1' \
    'djangorestframework-simplejwt==5.3.1' \
    'django-cors-headers==4.3.1' \
    'psycopg2-binary==2.9.9' \
    'channels==4.1.0' \
    'channels_redis==4.2.0' \
    'redis==4.6.0' \
    'daphne==4.1.2' \
    'gunicorn==22.0.0' \
    'whitenoise==6.7.0'"

# Если есть requirements.txt — используем его
if [ -f "requirements.txt" ]; then
    sudo -u ${REAL_USER} bash -c "source venv/bin/activate && pip install -r requirements.txt"
fi

echo -e "${GREEN}✅ Python зависимости установлены${NC}"

# ----------------------------------------------------------------------------
# ШАГ 8: Настройка Django
# ----------------------------------------------------------------------------
echo -e "${YELLOW}⚙️  Шаг 8/11: Настройка Django...${NC}"

cd ${PROJECT_DIR}

# Применяем миграции
sudo -u ${REAL_USER} bash -c "source venv/bin/activate && python manage.py migrate --noinput"

# Собираем статику
sudo -u ${REAL_USER} bash -c "source venv/bin/activate && python manage.py collectstatic --noinput"

# Создаём суперпользователя (если его нет)
sudo -u ${REAL_USER} bash -c "source venv/bin/activate && python manage.py shell << EOF
from django.contrib.auth import get_user_model
User = get_user_model()
if not User.objects.filter(username='${ADMIN_USERNAME}').exists():
    User.objects.create_superuser('${ADMIN_USERNAME}', '${ADMIN_EMAIL}', '${ADMIN_PASSWORD}')
    print('✅ Суперпользователь создан: ${ADMIN_USERNAME}')
else:
    print('ℹ️  Суперпользователь уже существует')
EOF"

echo -e "${GREEN}✅ Django настроен${NC}"

# ----------------------------------------------------------------------------
# ШАГ 9: Сборка Frontend (React + Vite)
# ----------------------------------------------------------------------------
echo -e "${YELLOW}⚛️  Шаг 9/11: Сборка Frontend...${NC}"

if [ -d "frontend" ]; then
    cd frontend
    
    # Устанавливаем зависимости
    sudo -u ${REAL_USER} npm install
    
    # Собираем продакшн версию
    sudo -u ${REAL_USER} npm run build
    
    cd ${PROJECT_DIR}
    echo -e "${GREEN}✅ Frontend собран в frontend/dist/${NC}"
else
    echo -e "${YELLOW}⚠️  Папка frontend не найдена${NC}"
fi

# ----------------------------------------------------------------------------
# ШАГ 10: Настройка Nginx
# ----------------------------------------------------------------------------
echo -e "${YELLOW}🌐 Шаг 10/11: Настройка Nginx...${NC}"

# Генерируем конфиг Nginx из шаблона
cat > /etc/nginx/sites-available/${DOMAIN} << EOF
server {
    listen 80;
    server_name ${DOMAIN} www.${DOMAIN};

    # Frontend (React) - раздаем напрямую через Nginx
    root ${PROJECT_DIR}/frontend/dist;
    index index.html;

    # Логи
    access_log /var/log/nginx/pos-access.log;
    error_log /var/log/nginx/pos-error.log;

    client_max_body_size 10M;

    # Статические файлы Django (CSS, JS, админка)
    location /static/ {
        alias ${PROJECT_DIR}/staticfiles/;
        expires 30d;
        add_header Cache-Control "public, immutable";
    }

    # Медиа файлы
    location /media/ {
        alias ${PROJECT_DIR}/media/;
        expires 7d;
    }

    # API Django (REST)
    location /api/ {
        proxy_pass http://127.0.0.1:8001;
        proxy_set_header Host \$host;
        proxy_set_header X-Real-IP \$remote_addr;
        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto \$scheme;
        proxy_read_timeout 300s;
    }

    # Django Admin
    location /admin/ {
        proxy_pass http://127.0.0.1:8001;
        proxy_set_header Host \$host;
        proxy_set_header X-Real-IP \$remote_addr;
        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto \$scheme;
    }

    # WebSocket для Django Channels
    location /ws/ {
        proxy_pass http://127.0.0.1:8001;
        proxy_http_version 1.1;
        proxy_set_header Upgrade \$http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host \$host;
        proxy_set_header X-Real-IP \$remote_addr;
        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto \$scheme;
        proxy_read_timeout 86400s;
    }

    # Frontend SPA — все остальные запросы
    location / {
        try_files \$uri \$uri/ /index.html;
    }
}
EOF

# Активируем сайт
ln -sf /etc/nginx/sites-available/${DOMAIN} /etc/nginx/sites-enabled/${DOMAIN}
rm -f /etc/nginx/sites-enabled/default

# Тестируем и перезапускаем Nginx
nginx -t
systemctl enable nginx
systemctl restart nginx

echo -e "${GREEN}✅ Nginx настроен${NC}"

# ----------------------------------------------------------------------------
# ШАГ 11: Настройка Supervisor
# ----------------------------------------------------------------------------
echo -e "${YELLOW}🔄 Шаг 11/11: Настройка Supervisor...${NC}"

# Конфиг для Daphne (WebSocket + HTTP)
cat > /etc/supervisor/conf.d/pos-daphne.conf << EOF
[program:pos-daphne]
command=${PROJECT_DIR}/venv/bin/daphne -b 127.0.0.1 -p 8001 core.asgi:application
directory=${PROJECT_DIR}
user=${REAL_USER}
autostart=true
autorestart=true
redirect_stderr=true
stdout_logfile=/var/log/pos-daphne.log
environment=DJANGO_SETTINGS_MODULE="core.settings"
EOF

# Перезапускаем Supervisor
systemctl enable supervisor
supervisorctl reread
supervisorctl update
supervisorctl restart pos-daphne

echo -e "${GREEN}✅ Supervisor настроен${NC}"

# ----------------------------------------------------------------------------
# Настройка Firewall
# ----------------------------------------------------------------------------
echo -e "${YELLOW}🔒 Настройка Firewall...${NC}"
ufw allow OpenSSH
ufw allow 'Nginx Full'
ufw --force enable

echo -e "${GREEN}✅ Firewall настроен${NC}"

# ----------------------------------------------------------------------------
# Финальная проверка
# ----------------------------------------------------------------------------
echo ""
echo -e "${GREEN}"
echo "╔════════════════════════════════════════════════════════════╗"
echo "║   ✅ РАЗВЕРТЫВАНИЕ ЗАВЕРШЕНО УСПЕШНО!                     ║"
echo "╚════════════════════════════════════════════════════════════╝"
echo -e "${NC}"

echo -e "${BLUE}📊 Статус сервисов:${NC}"
echo "  • PostgreSQL: $(systemctl is-active postgresql)"
echo "  • Redis:      $(systemctl is-active redis-server)"
echo "  • Nginx:      $(systemctl is-active nginx)"
echo "  • Daphne:     $(supervisorctl status pos-daphne | awk '{print $2}')"
echo ""
echo -e "${BLUE}🌐 Ваш сайт:${NC}"
echo "  http://${DOMAIN}"
echo ""
echo -e "${BLUE}🔐 Django Admin:${NC}"
echo "  http://${DOMAIN}/admin/"
echo "  👤 Логин: ${ADMIN_USERNAME}"
echo "  🔑 Пароль: ${ADMIN_PASSWORD}"
echo ""
echo -e "${BLUE}📝 Логи:${NC}"
echo "  • Daphne: sudo tail -f /var/log/pos-daphne.log"
echo "  • Nginx:  sudo tail -f /var/log/nginx/pos-error.log"
echo ""
echo -e "${YELLOW}⚠️  Следующие шаги:${NC}"
echo "  1. Настроить DNS для домена ${DOMAIN}"
echo "  2. Установить SSL: sudo certbot --nginx -d ${DOMAIN}"
echo "  3. Обновить ALLOWED_HOSTS в core/settings.py"
echo ""
echo -e "${GREEN}🚀 Готово к работе!${NC}"
```

Сделай скрипт исполняемым:

## 🚀 Как использовать

### На локальной машине:

```bash
# 1. Положи все 4 файла в корень проекта
cd ~/pos-system
# (файлы уже должны быть там)

# 2. Создай архив, исключая мусор
cd ~
tar -czf pos-deploy.tar.gz \
    --exclude='pos-system/venv' \
    --exclude='pos-system/frontend/node_modules' \
    --exclude='pos-system/__pycache__' \
    --exclude='pos-system/**/__pycache__' \
    --exclude='pos-system/*.sqlite3' \
    --exclude='pos-system/staticfiles' \
    pos-system/

# 3. Загрузи на сервер
scp pos-deploy.tar.gz mike@your-server-ip:/tmp/

# 4. Подключись к серверу
ssh mike@your-server-ip
```

### На сервере:

```bash
# 1. Распакуй архив
cd /tmp
tar -xzf pos-deploy.tar.gz
cd pos-system

# 2. Настрой .env
cp .env.example .env
nano .env  # Заполни все поля!

# 3. Запусти деплой
sudo ./deploy.sh
```

## 🔧 Важные особенности:

- **Фиксированные версии** — все пакеты в `requirements.txt` с точными версиями, чтобы избежать сюрпризов
- **`redis==4.6.0`** — специально зафиксирована, чтобы избежать бага с RESP3, который мы чинили ранее
- **Идемпотентность** — скрипт можно запускать повторно, он не сломает существующую БД
- **Безопасность** — пароли в `.env`, а не в коде
- **Логирование** — все логи в `/var/log/pos-*.log`
